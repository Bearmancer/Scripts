**Post-modification code chunk for `Up()` and `Down()` (inserting custom index scripts):**
```csharp
	protected override void Up(MigrationBuilder migrationBuilder)
	{
		migrationBuilder.Sql("CREATE EXTENSION IF NOT EXISTS unaccent;");
		migrationBuilder.Sql("CREATE EXTENSION IF NOT EXISTS pg_trgm;");

		// Custom GIN, BRIN, and Functional Indexes
		migrationBuilder.Sql("CREATE UNIQUE INDEX IF NOT EXISTS ix_artists_name_normalized ON \"artists\" (lower(unaccent(\"Name\")));");
		migrationBuilder.Sql("CREATE INDEX IF NOT EXISTS ix_artists_name_trgm ON \"artists\" USING gin (\"Name\" gin_trgm_ops);");
		migrationBuilder.Sql("CREATE INDEX IF NOT EXISTS ix_albums_title_trgm ON \"albums\" USING gin (\"Title\" gin_trgm_ops);");
		migrationBuilder.Sql("CREATE INDEX IF NOT EXISTS ix_tracks_title_trgm ON \"tracks\" USING gin (\"Title\" gin_trgm_ops);");
		migrationBuilder.Sql("CREATE INDEX IF NOT EXISTS ix_videos_title_trgm ON \"videos\" USING gin (\"Title\" gin_trgm_ops);");
		migrationBuilder.Sql("CREATE INDEX IF NOT EXISTS ix_scrobbles_scrobbled_at_brin ON \"scrobbles\" USING brin (\"ScrobbledAt\");");
```
```csharp
	protected override void Down(MigrationBuilder migrationBuilder)
	{
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_scrobbles_scrobbled_at_brin;");
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_videos_title_trgm;");
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_tracks_title_trgm;");
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_albums_title_trgm;");
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_artists_name_trgm;");
		migrationBuilder.Sql("DROP INDEX IF EXISTS ix_artists_name_normalized;");

		migrationBuilder.Sql("DROP EXTENSION IF EXISTS pg_trgm;");
		migrationBuilder.Sql("DROP EXTENSION IF EXISTS unaccent;");
```
